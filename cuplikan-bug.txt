namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\FotoProduk;

class FotoProdukController extends Controller
{
    public function store(Request $request)
    {
        $request->validate([
            'foto_produk' => 'required|image|mimes:jpeg,jpg,png,gif|max:1024',
            'produk_id'   => 'required|exists:produk,id',
        ]);

        if ($request->file('foto_produk')) {
            $file = $request->file('foto_produk');
            $extension = $file->getClientOriginalExtension();
            $fileName = date('YmdHis') . '_' . uniqid() . '.' . $extension;

            $file->move(public_path('storage/img-produk/'), $fileName);

            FotoProduk::create([
                'produk_id' => $request->produk_id,
                'foto'      => $fileName,
            ]);
        }

        return redirect()->back()->with('success', 'Foto berhasil ditambahkan');
    }

    public function destroy(string $id)
    {
        $foto = FotoProduk::findOrFail($id);
        $fotoPath = public_path('storage/img-produk/' . $foto->foto);

        if ($foto->foto && file_exists($fotoPath)) {
            unlink($fotoPath);
        }

        $foto->delete();

        return redirect()->back()->with('success', 'Foto berhasil dihapus');
    }
}

<?php

    {
        $kategori = Kategori::orderBy('nama_kategori', 'asc')->get();

            'judul'    => 'Tambah Produk',
            'kategori' => $kategori,

    public function store(Request $request)
    {
        $validatedData = $request->validate([
            'kategori_id' => 'required',
            'nama_produk' => 'required|max:255|unique:produk',
            'detail'      => 'required',
            'harga'       => 'required',
            'berat'       => 'required',
            'stok'        => 'required',
            'foto'        => 'required|image|mimes:jpeg,jpg,png,gif|file|max:1024',
        ], [
            'foto.image' => 'Format gambar gunakan file dengan ekstensi jpeg, jpg, png, atau gif.',
            'foto.max'   => 'Ukuran file gambar Maksimal adalah 1024 KB.',
        ]);

        $validatedData['user_id'] = auth()->id();
        $validatedData['status']  = 0;

        if ($request->file('foto')) {
            $file             = $request->file('foto');
            $extension        = $file->getClientOriginalExtension();
            $originalFileName = date('YmdHis') . '_' . uniqid() . '.' . $extension;
            $directory        = 'storage/img-produk/';

            ImageHelper::uploadAndResize($file, $directory, $originalFileName);

            $thumbnailLg = 'thumb_lg_' . $originalFileName;
            ImageHelper::uploadAndResize($file, $directory, $thumbnailLg, 800, null);

            $thumbnailMd = 'thumb_md_' . $originalFileName;
            ImageHelper::uploadAndResize($file, $directory, $thumbnailMd, 500, 519);

            $thumbnailSm = 'thumb_sm_' . $originalFileName;
            ImageHelper::uploadAndResize($file, $directory, $thumbnailSm, 100, 110);

            $validatedData['foto'] = $originalFileName;
        }

        $produk = Produk::create($validatedData);

        // Simpan juga ke tabel foto_produk
        if (!empty($validatedData['foto'])) {
            FotoProduk::create([
                'produk_id' => $produk->id,
                'foto'      => $validatedData['foto'],
            ]);
        }

        return redirect()->route('backend.produk.index')
            ->with('success', 'Data berhasil tersimpan');

    {
        $produk = Produk::findOrFail($id);

        $rules = [
            'nama_produk' => 'required|max:255|unique:produk,nama_produk,' . $id,
            'kategori_id' => 'required',
            'status'      => 'required',
            'detail'      => 'required',
            'harga'       => 'required',

            'stok'        => 'required',

        $validatedData = $request->validate($rules, $messages);
        $validatedData['user_id'] = auth()->id();

        if ($request->file('foto')) {
            if ($produk->foto) {
                $directory = public_path('storage/img-produk/');

                $oldImagePath = $directory . $produk->foto;
                if (file_exists($oldImagePath)) unlink($oldImagePath);

                $oldThumbnailLg = $directory . 'thumb_lg_' . $produk->foto;
                if (file_exists($oldThumbnailLg)) unlink($oldThumbnailLg);

                $oldThumbnailMd = $directory . 'thumb_md_' . $produk->foto;
                if (file_exists($oldThumbnailMd)) unlink($oldThumbnailMd);

                $oldThumbnailSm = $directory . 'thumb_sm_' . $produk->foto;
                if (file_exists($oldThumbnailSm)) unlink($oldThumbnailSm);
            }

            $file             = $request->file('foto');
            $extension        = $file->getClientOriginalExtension();
            $originalFileName = date('YmdHis') . '_' . uniqid() . '.' . $extension;
            $directory        = 'storage/img-produk/';

            ImageHelper::uploadAndResize($file, $directory, $originalFileName);

            $thumbnailLg = 'thumb_lg_' . $originalFileName;
            ImageHelper::uploadAndResize($file, $directory, $thumbnailLg, 800, null);

            $thumbnailMd = 'thumb_md_' . $originalFileName;
            ImageHelper::uploadAndResize($file, $directory, $thumbnailMd, 500, 519);

            $thumbnailSm = 'thumb_sm_' . $originalFileName;
            ImageHelper::uploadAndResize($file, $directory, $thumbnailSm, 100, 110);

            FotoProduk::where('produk_id', $id)->delete();
            FotoProduk::create([
                'produk_id' => $id,
                'foto'      => $originalFileName,

    public function destroy($id)
    {
        $produk    = Produk::findOrFail($id);
        $directory = public_path('storage/img-produk/');

        if ($produk->foto) {
            $oldImagePath = $directory . $produk->foto;
            if (file_exists($oldImagePath)) unlink($oldImagePath);

            $thumbnailLg = $directory . 'thumb_lg_' . $produk->foto;
            if (file_exists($thumbnailLg)) unlink($thumbnailLg);

            $thumbnailMd = $directory . 'thumb_md_' . $produk->foto;
            if (file_exists($thumbnailMd)) unlink($thumbnailMd);

    public function berandaBackend()
    {
        $user = auth()->user();

        // ===== STATISTIK UMUM =====
        $totalPesanan    = Pesanan::count();
        $totalPendapatan = Pesanan::where('status', 'selesai')->sum('total_harga');

        // ===== PESANAN PENDING =====
        $pesananPending = Pesanan::where('status', 'menunggu pembayaran')->count();

        // ===== STOK MENIPIS COUNT (untuk kartu) =====
        $stokMenipis = Produk::where('stok', '<', 5)->count();

        // ===== STOK MENIPIS LIST (untuk tabel warning) =====
        $stokMenipisList = Produk::where('stok', '<', 5)
            ->with('kategori')
            ->orderBy('stok', 'asc')
            ->get();

        // ===== GRAFIK PESANAN 6 BULAN TERAKHIR (render pertama kali) =====
        [$grafikLabel, $grafikData] = $this->hitungGrafikPesanan();

        // ===== PESANAN TERBARU =====

        // ===== PRODUK TERLARIS =====
        $produkTerlaris = Produk::withCount('pesananDetail')
            ->orderBy('pesanan_detail_count', 'desc')
            ->take(5)
            ->get();

        return view('backend.v_beranda.index', compact(
            'totalPesanan',
            'totalPendapatan',
            'totalProduk',
            'totalCustomer',
            'pesananHariIni',
            'pesananPending',
            'stokMenipis',
            'stokMenipisList',
            'grafikData',
            'grafikLabel',
            'pesananTerbaru',
            'produkTerlaris',
        ));
    }

    /**
     * Endpoint AJAX untuk update grafik pesanan secara berkala (real-time polling).
     * Dipanggil dari JS dashboard tiap beberapa detik.
     */
    public function grafikPesananAjax()
    {
        [$grafikLabel, $grafikData] = $this->hitungGrafikPesanan();

        return response()->json([
            'labels' => $grafikLabel,
            'data'   => $grafikData,
            'jam'    => Carbon::now()->format('H:i:s'),
        ]);
    }

    /**
     * Hitung ulang data grafik pesanan 6 bulan terakhir.
     * Dipakai bareng oleh berandaBackend() dan grafikPesananAjax()
     * supaya query-nya selalu konsisten dan datanya selalu yang terbaru.
     */
    private function hitungGrafikPesanan(): array
    {
        $grafikData  = [];
        $grafikLabel = [];

        for ($i = 5; $i >= 0; $i--) {
            $bulan         = Carbon::now()->subMonths($i);
            $grafikLabel[] = $bulan->translatedFormat('M Y');
            $grafikData[]  = Pesanan::whereYear('created_at', $bulan->year)
                ->whereMonth('created_at', $bulan->month)
                ->count();
        }

        return [$grafikLabel, $grafikData];
    }

    public function index()
    {
        $produk = Produk::where('status', 1)->orderBy('updated_at', 'desc')->paginate(6);
        $kategoris = Kategori::all();
        $produkBanner1 = Produk::whereHas('kategori', fn($q) => $q->where('nama_kategori', 'like', '%Tas%'))->first() ?? Produk::first();
        $produkBanner2 = Produk::whereHas('kategori', fn($q) => $q->where('nama_kategori', 'like', '%Pakaian%'))->first() ?? Produk::first();
        $produkBanner3 = Produk::orderBy('created_at', 'desc')->first();
        return view('v_beranda.index', [
            'judul' => 'Halaman Beranda',
            'produk' => $produk,
            'kategoris' => $kategoris,
            'produkBanner1' => $produkBanner1,
            'produkBanner2' => $produkBanner2,
            'produkBanner3' => $produkBanner3,
        ]);
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use App\Models\Keranjang;
use App\Models\Produk;
use App\Models\Pesanan;
use App\Models\PesananDetail;
use App\Models\User;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Mail;
use App\Mail\StokMenipisMail;

class KeranjangController extends Controller
{
    public function index()
    {
        $keranjangs = Keranjang::with('produk')
            ->where('user_id', Auth::id())
            ->get();

        return view('keranjang.index', compact('keranjangs'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'produk_id' => 'required|exists:produk,id',
            'jumlah'    => 'required|integer|min:1',
        ]);

        $produk    = Produk::findOrFail($request->produk_id);
        $keranjang = Keranjang::where('user_id', Auth::id())
            ->where('produk_id', $request->produk_id)
            ->first();

        if ($keranjang) {
            $keranjang->update([
                'jumlah' => $keranjang->jumlah + $request->jumlah
            ]);
        } else {
            Keranjang::create([
                'user_id'   => Auth::id(),
                'produk_id' => $request->produk_id,
                'jumlah'    => $request->jumlah,
                'harga'     => $produk->harga,
            ]);
        }

        if ($request->input('redirect') === 'keranjang') {
            return redirect()
                ->route('customer.keranjang.index')
                ->with('success', 'Produk "' . $produk->nama_produk . '" berhasil ditambahkan ke keranjang!');
        }

        return redirect()->back()
            ->with('success', 'Produk "' . $produk->nama_produk . '" berhasil ditambahkan ke keranjang!');
    }

    public function update(Request $request, $id)
    {
        $request->validate([
            'jumlah' => 'required|integer|min:1'
        ]);

        $keranjang = Keranjang::where('user_id', Auth::id())
            ->where('id', $id)
            ->firstOrFail();

        $keranjang->update(['jumlah' => $request->jumlah]);

        return redirect()->back()->with('success', 'Keranjang berhasil diupdate!');
    }

    public function destroy($id)
    {
        $keranjang = Keranjang::where('user_id', Auth::id())
            ->where('id', $id)
            ->firstOrFail();

        $keranjang->delete();

        return redirect()->back()->with('success', 'Produk berhasil dihapus dari keranjang!');
    }

    public function clear()
    {
        Keranjang::where('user_id', Auth::id())->delete();

        return redirect()->back()->with('success', 'Keranjang berhasil dikosongkan!');
    }

    public function count()
    {
        $count = Keranjang::where('user_id', Auth::id())->sum('jumlah');

        return response()->json(['count' => $count]);
    }

    public function checkout()
    {
        $keranjangs = Keranjang::with('produk')
            ->where('user_id', Auth::id())
            ->get();

        if ($keranjangs->isEmpty()) {
            return redirect()
                ->route('customer.keranjang.index')
                ->with('error', 'Keranjang masih kosong!');
        }

        $total = $keranjangs->sum('subtotal');

        return view('keranjang.checkout', compact('keranjangs', 'total'));
    }

    public function prosesCheckout(Request $request)
    {
        $keranjangs = Keranjang::with('produk')
            ->where('user_id', Auth::id())
            ->get();

        if ($keranjangs->isEmpty()) {
            return redirect()
                ->route('customer.keranjang.index')
                ->with('error', 'Keranjang masih kosong!');
        }

        // Cek stok semua produk
        foreach ($keranjangs as $item) {
            if ($item->produk->stok < $item->jumlah) {
                return redirect()
                    ->route('customer.keranjang.index')
                    ->with('error', 'Stok produk "' . $item->produk->nama_produk . '" tidak mencukupi! Sisa stok: ' . $item->produk->stok);
            }
        }

        $total = $keranjangs->sum('subtotal');

        // Simpan ke tabel pesanan
        $pesanan = Pesanan::create([
            'user_id'      => Auth::id(),
            'kode_pesanan' => 'ORD-' . strtoupper(uniqid()),
            'total_harga'  => $total,
            'status'       => 'Menunggu Pembayaran',
        ]);

        // Simpan detail pesanan dan kurangi stok
        foreach ($keranjangs as $item) {
            PesananDetail::create([
                'pesanan_id'  => $pesanan->id,
                'produk_id'   => $item->produk_id,
                'nama_produk' => $item->produk->nama_produk,
                'harga'       => $item->harga,
                'jumlah'      => $item->jumlah,
                'subtotal'    => $item->subtotal,
            ]);

            $item->produk->decrement('stok', $item->jumlah);
        }

        // Kosongkan keranjang
        Keranjang::where('user_id', Auth::id())->delete();

        // Cek stok menipis dan kirim email ke admin
        $produkMenipis = Produk::where('stok', '<', 5)->with('kategori')->get();
        if ($produkMenipis->isNotEmpty()) {
            $admins = User::whereIn('role', ['0', '1'])->get();
            foreach ($admins as $admin) {
                try {
                    Mail::to($admin->email)->send(new StokMenipisMail($produkMenipis));
                } catch (\Exception $e) {
                    // Gagal kirim email tidak mengganggu proses checkout
                }
            }
        }

        // â† Redirect ke halaman pembayaran (BUKAN riwayat)
        return redirect()->route('customer.checkout.pembayaran', $pesanan->id);
    }

    public function riwayatBelanja()

        $pesanan = Pesanan::with('details')
            ->where('user_id', auth()->id())
            ->orderBy('created_at', 'desc')
            ->get();

        return view('customer.riwayat', compact('pesanan'));
    }
}

    public function addToCart(Request $request, $id)
    {
        $customer = $this->customerSaatIni();
        if (!$customer) {
            return $this->bukanCustomer();
        }

        $produk = Produk::findOrFail($id);

        $jumlahDitambah = max(1, (int) $request->input('jumlah', 1));

        $order = Order::firstOrCreate(
            ['customer_id' => $customer->id, 'status' => 'pending'],
            ['total_harga' => 0]
        );

        $orderItem = OrderItem::firstOrCreate(
            ['order_id' => $order->id, 'produk_id' => $produk->id],
            [
                'nama_produk' => $produk->nama_produk,
                'harga'       => $produk->harga,
                'jumlah'      => $jumlahDitambah,
                'subtotal'    => $produk->harga * $jumlahDitambah,
            ]
        );

        if (!$orderItem->wasRecentlyCreated) {
            $orderItem->jumlah += $jumlahDitambah;
            $orderItem->subtotal = $orderItem->jumlah * $orderItem->harga;
            $orderItem->save();
        }

        $order->total_harga = $order->orderItems()->sum('subtotal');
        $order->save();

        return redirect()->route('order.cart')->with('success', 'Produk berhasil ditambahkan ke keranjang');
    }

    public function selectPayment($id)
    {
        $pesanan = Pesanan::with('detailPesanan')
            ->where('user_id', Auth::id())
            ->findOrFail($id);

        // Sudah dibayar / diproses / dibatalkan -> jangan bikin token baru
        if ($pesanan->status !== 'Menunggu Pembayaran') {
            return redirect()->route('customer.checkout.selesai', $pesanan->id);
        }

        // Token sudah ada (dan punya order_id Midtrans-nya) -> pakai lagi
        if ($pesanan->snap_token && $pesanan->midtrans_order_id) {
            return view('customer.checkout.pembayaran', [
                'pesanan'   => $pesanan,
                'snapToken' => $pesanan->snap_token,
            ]);
        }

        Config::$serverKey    = config('midtrans.server_key');
        Config::$isProduction = config('midtrans.is_production');
        Config::$isSanitized  = true;
        Config::$is3ds        = true;

        // order_id unik tiap kali token dibuat (Midtrans mengunci order_id begitu token terbentuk)
        $midtransOrderId = $pesanan->kode_pesanan . '-' . time();

        $params = [
            'transaction_details' => [
                'order_id'     => $midtransOrderId,
                'gross_amount' => (int) $pesanan->total_harga,
            ],
            'customer_details' => [
                'first_name' => Auth::user()->nama,
                'email'      => Auth::user()->email,
            ],
            'callbacks' => [
                'finish' => route('customer.checkout.selesaiMidtrans'),
            ],
        ];

        $snapToken = Snap::getSnapToken($params);

        $pesanan->snap_token        = $snapToken;
        $pesanan->midtrans_order_id = $midtransOrderId;
        $pesanan->save();

        return view('customer.checkout.pembayaran', [
            'pesanan'   => $pesanan,
            'snapToken' => $snapToken,
        ]);
    }

    /**
     * Menerima notifikasi pembayaran dari Midtrans (webhook).
     * Route ini TIDAK boleh dibungkus middleware auth, karena
     * yang memanggil adalah server Midtrans, bukan user yang login.
     */
    public function callback(Request $request)
    {
        $serverKey = config('midtrans.server_key');
        $hashed = hash('sha512', $request->order_id . $request->status_code . $request->gross_amount . $serverKey);

        if ($hashed == $request->signature_key) {
            $pesanan = Pesanan::where('midtrans_order_id', $request->order_id)->orWhere('kode_pesanan', $request->order_id)->first();

            if ($pesanan && in_array($pesanan->status, ['Menunggu Pembayaran', 'Paid'])) {
                if ($request->transaction_status == 'settlement' || $request->transaction_status == 'capture') {
                    $pesanan->update(['status' => 'Proses']);
                } elseif ($request->transaction_status == 'expire') {
                    $pesanan->update(['status' => 'Kadaluarsa']);
                } elseif ($request->transaction_status == 'cancel' || $request->transaction_status == 'deny') {
                    $pesanan->update(['status' => 'Dibatalkan']);
                }
            }
        }

        return response()->json(['message' => 'OK']);
    }

    /**
     * Dipanggil saat customer kembali dari Midtrans. Status dicek langsung ke API Midtrans
     * lalu pesanan diubah ke "Proses" kalau pembayaran sukses.
     */
    public function selesaiMidtrans(Request $request)
    {
        $orderId = $request->query('order_id');

        $pesanan = Pesanan::where('user_id', Auth::id())
            ->where(function ($q) use ($orderId) {
                $q->where('midtrans_order_id', $orderId)
                  ->orWhere('kode_pesanan', $orderId);
            })
            ->firstOrFail();

        $idMidtrans = $pesanan->midtrans_order_id ?: $pesanan->kode_pesanan;

        Config::$serverKey    = config('midtrans.server_key');
        Config::$isProduction = config('midtrans.is_production');

        try {
            $status = \Midtrans\Transaction::status($idMidtrans);
            $trx    = $status->transaction_status ?? null;
            $fraud  = $status->fraud_status ?? null;

            if (in_array($pesanan->status, ['Menunggu Pembayaran', 'Paid'])) {
                if ($trx === 'settlement' || ($trx === 'capture' && $fraud !== 'challenge')) {
                    $pesanan->update(['status' => 'Proses']);
                } elseif ($trx === 'expire') {
                    $pesanan->update(['status' => 'Kadaluarsa']);
                } elseif ($trx === 'cancel' || $trx === 'deny') {
                    $pesanan->update(['status' => 'Dibatalkan']);
                }
            }
        } catch (\Throwable $e) {
            \Log::warning('Gagal cek status Midtrans untuk ' . $idMidtrans . ': ' . $e->getMessage());
        }

        return redirect()->route('customer.checkout.selesai', $pesanan->id);
    }

<script>
document.addEventListener('DOMContentLoaded', function() {

    const btnTambah = document.getElementById('btn-tambah-foto');
    const formTambah = document.getElementById('form-tambah-foto');
    const btnBatal = document.getElementById('btn-batal');

    btnTambah.addEventListener('click', function() {
        formTambah.style.display = 'block';
        btnTambah.style.display = 'none';
        formTambah.scrollIntoView({
            behavior: 'smooth'
        });
    });

    btnBatal.addEventListener('click', function() {
        formTambah.style.display = 'none';
        btnTambah.style.display = 'block';
        document.getElementById('form-foto-store').querySelector('input[type="file"]').value = '';
    });

});

function hapusFoto(id) {
    if (!confirm('Yakin ingin menghapus foto ini?')) return;

    fetch(`/foto-produk/${id}`, {
            method: 'DELETE',
            headers: {
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').getAttribute('content'),
                'Accept': 'application/json',
            }
        })
        .then(res => {
            if (res.ok) {
                const row = document.getElementById('foto-row-' + id);
                if (row) row.remove();
            } else {
                alert('Gagal menghapus foto. Silakan coba lagi.');
            }
        })
        .catch(() => alert('Terjadi kesalahan jaringan.'));
}
