    public function simpanUlasan(Request $request, $id)
    {
        $request->validate([
            'produk_id' => 'required|integer',
            'rating'    => 'required|integer|min:1|max:5',
            'komentar'  => 'required|string|min:10|max:1000',
            'foto'      => 'nullable|image|mimes:jpg,jpeg,png,webp|max:3072',
        ], [
            'rating.required' => 'Pilih jumlah bintang dulu.',
            'komentar.required' => 'Tulis ulasanmu dulu.',
            'komentar.min'      => 'Ulasan minimal 10 karakter.',
            'rating.min'      => 'Pilih jumlah bintang dulu.',
            'foto.image'      => 'File harus berupa gambar.',
            'foto.max'        => 'Ukuran foto maksimal 3 MB.',
        ]);

        $pesanan = Pesanan::with('details')
            ->where('id', $id)
            ->where('user_id', Auth::id())
            ->firstOrFail();

        if (strtolower(trim((string) $pesanan->status)) !== 'selesai') {
            return response()->json(['message' => 'Ulasan hanya bisa diberikan setelah pesanan selesai.'], 422);
        }

        $produkId = (int) $request->produk_id;

        if (!empty($pesanan->rating_logistik)) {
            return response()->json(['message' => 'Pesanan ini sudah dinilai lewat Lacak, jadi tidak bisa diulas lagi.'], 422);
        }

        if (!$pesanan->details->contains('produk_id', $produkId)) {
            return response()->json(['message' => 'Produk tidak ditemukan di pesanan ini.'], 422);
        }

        $sudah = ProductReview::where('user_id', Auth::id())
            ->where('pesanan_id', $pesanan->id)
            ->exists();

        if ($sudah) {
            return response()->json(['message' => 'Produk ini sudah kamu ulas.'], 422);
        }

        $path = $request->hasFile('foto')
            ? $request->file('foto')->store('ulasan', 'public')
            : null;

        ProductReview::create([
            'produk_id'  => $produkId,
            'user_id'    => Auth::id(),
            'pesanan_id' => $pesanan->id,
            'rating'     => (int) $request->rating,
            'komentar'   => $request->komentar,
            'foto'       => $path,
        ]);

        return response()->json(['ok' => true]);
    }
