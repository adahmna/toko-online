/* chat-mic-kirim */
(function () {
    var input, send, mic;

    function cari() {
        input = document.querySelector('input[placeholder^="Ketik pesan"], textarea[placeholder^="Ketik pesan"]');
        if (!input) return false;
        var p = input;
        for (var i = 0; i < 6 && p.parentElement; i++) {
            p = p.parentElement;
            var els = p.querySelectorAll('button, a, div[role="button"], span[role="button"], div, span');
            for (var j = 0; j < els.length; j++) {
                var el = els[j];
                var t = (el.textContent || '').trim().toLowerCase();
                var ident = ((el.id || '') + ' ' + (typeof el.className === 'string' ? el.className : '')).toLowerCase();
                if (!send && t === 'kirim' && el.children.length === 0) send = el;
                if (!mic && /mic|voice|suara/.test(ident)) mic = el;
            }
            if (send && mic) break;
        }
        if (send) { var b = send.closest('button, a'); if (b) send = b; }
        return !!(send && mic);
    }

    function tampil(el, ya) {
        if (ya) el.style.removeProperty('display');
        else el.style.setProperty('display', 'none', 'important');
    }

    function cek() {
        var ada = input.value.trim().length > 0;
        tampil(send, ada);
        tampil(mic, !ada);
    }

    var coba = 0;
    var tunggu = setInterval(function () {
        coba++;
        if (cari()) {
            clearInterval(tunggu);
            input.addEventListener('input', cek);
            input.addEventListener('keyup', cek);
            setInterval(cek, 300); // menangkap input yang dikosongkan lewat JS setelah kirim
            cek();
        } else if (coba > 40) {
            clearInterval(tunggu);
            console.warn('chat-mic-kirim: elemen chat tidak ditemukan');
        }
    }, 250);
})();
</script>
